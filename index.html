function iterAdd(iter, amountSci) { const a = iterAsSci(iter); const b = toBig(amountSci); const aHuge = Math.abs(a.e) > MAX_SAFE_EXP; const bHuge = Math.abs(b.e) > MAX_SAFE_EXP; let result; if (aHuge && !bHuge) { // a has a huge exponent (Googolplex-class). Absorb b into a's mantissa // so that +1 accumulates visibly instead of being swallowed by the // bigAdd dominant-term shortcut. if (b.e >= 0 && b.e < MAX_ABSORB_EXP) { const bMant = b.m * (10n ** BigInt(Math.round(b.e))); result = { m: a.m + bMant, e: a.e }; } else { result = bigAdd(a, b); } } else if (bHuge && !aHuge) { // b dominates — a is negligible result = b; } else if (aHuge && bHuge) { // Both huge — add mantissas only if exponents match closely const expDiff = Math.abs(a.e - b.e); const expScale = Math.max(Math.abs(a.e), 1); if (expDiff / expScale < 1e-6) { result = { m: a.m + b.m, e: a.e }; } else { result = a; } } else { result = bigAdd(a, b); } // Downgrade to Number if safely small if (typeof iter === "number" && result.e >= 0 && result.e < 15) { const n = Number(result.m) * Math.pow(10, result.e); if (Number.isSafeInteger(n) && Math.abs(n) < 1e15) return n; } return result; }
